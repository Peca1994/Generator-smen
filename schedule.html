const db = getFirestore(app); /* DOM */ const dateEl = document.getElementById("date"); const shiftEl = document.getElementById("shift"); const nameFilterEl = document.getElementById("nameFilter"); const gridEl = document.getElementById("grid"); const shiftInfoBox = document.getElementById("shiftInfoBox"); const infoSK = document.getElementById("infoSK"); const infoFA = document.getElementById("infoFA"); const infoZA = document.getElementById("infoZA"); dateEl.value = new Date().toISOString().slice(0,10); /* POZICE */ const POSITIONS = [ "DP-L","DP-P","ZRC-L","ZRC-P","T-L","T-P", "AGR-LP","AGR-PP","AGR-PZ","AGR-LZ", "MOT-L","MOT-P","LEP-LP","LEP-PP","LEP-Z", "REPRO-L","REPRO-P","VYP1-L","VYP1-P", "VYP2-L","VYP2-P","PR-L","PR-P","ZAM-L","ZAM-P","VET","VET2","INST", "INST2","JINE","JINE2", ]; let workerMap = {}; // id -> name let phasesData = []; /* LOAD WORKERS MAP */ async function loadWorkers(){ const snap = await getDocs(collection(db,"employees")); workerMap = {}; snap.forEach(d=>{ const e=d.data(); workerMap[d.id] = e.name; }); } /* TIME LABELS */ function phaseLabels(){ const s=shiftEl.value; if(s==="ranni") return ["6:00-8:00","8:00-10:15","10:45-12:00","12:00-14:00"]; if(s==="odpo") return ["14:00-16:00","16:00-18:15","18:45-20:00","20:00-22:00"]; return ["22:00-00:00","00:00-02:15","02:45-04:00","04:00-06:00"]; } /* LOAD SHIFT */ async function loadShift(){ const id = dateEl.value + "_" + shiftEl.value; const snap = await getDoc(doc(db,"shifts",id)); if(!snap.exists()){ gridEl.innerHTML = "Směna nenalezena"; return; } const data = snap.data(); phasesData = data.phases; // 🔷 info o směně pro usera renderShiftInfo(data.info); render(); } /* RENDER */ function render(){ const filter = nameFilterEl.value.toLowerCase().trim(); const labels = phaseLabels(); let html = "Pozice"; labels.forEach(l=> html+=`${l}`); html+=""; POSITIONS.forEach(pos=>{ const names = phasesData.map(ph=>{ const wid = ph[pos]; return wid ? workerMap[wid] || "?" : ""; }); // filtr — zobraz jen když tam ten člověk někde je if(filter){ if(!names.some(n=>n.toLowerCase().includes(filter))){ return; } } html += `${pos}`; names.forEach(n=>{ if(!n){ html+=`—`; }else if(filter && n.toLowerCase().includes(filter)){ html+=`${n}`; }else{ html+=`${n}`; } }); html+=""; }); gridEl.innerHTML = html; } function renderShiftInfo(info){ if(!info){ shiftInfoBox.style.display = "none"; return; } shiftInfoBox.style.display = "block"; infoSK.innerHTML = `
Scala/Kamiq${info.sk?.start||""} → ${info.sk?.end||""}
`; infoFA.innerHTML = `
Fabie${info.fa?.start||""} → ${info.fa?.end||""}
`; infoZA.innerHTML = `
Zásobník${info.za?.start||""} → ${info.za?.end||""}
`; } /* EVENTS */ dateEl.onchange = loadShift; shiftEl.onchange = loadShift; nameFilterEl.oninput = render; /* INIT */ await loadWorkers(); await loadShift();
